# this code is a bit round-about, but it works.

set_table <- function(df, var_name){
  df_group <- df %>% 
    group_by({{var_name}}) %>%
    summarize(k_articles = n(), n_studies = sum(denom, na.rm = T)) %>%
    arrange(desc(k_articles), desc(n_studies)) %>% 
    rename(var1 = {{var_name}})
  return(df_group)
}
