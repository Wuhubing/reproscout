# create a function to output the power and sensitivity of each study included in the reviews.
output <- function(d){
  
for (i in 1:nrow(d)){
  
# calculate power for small, medium, and large cohen's effect sizes, using d = f*2. Also calculate sensitivity for power = 0.8 and 0.95. The type of anova to run is coded in the data, where type = 2 is for interactions, and type = 1 is for between groups. 
  if (d$test[i] == "anova"){
    d$small[i] <- wp.rmanova(d$n[i], d$ng[i], d$nm[i], f = 0.1, nscor = 1, alpha = 0.05, power = NULL, type = d$type[i])$power
    d$medium[i] <- wp.rmanova(d$n[i], d$ng[i], d$nm[i], f = 0.25, nscor = 1, alpha = 0.05, power = NULL, type = d$type[i])$power
    d$large[i] <- wp.rmanova(d$n[i], d$ng[i], d$nm[i], f = 0.4, nscor = 1, alpha = 0.05, power = NULL, type = d$type[i])$power
    d$p80[i] <- wp.rmanova(d$n[i], d$ng[i], d$nm[i], f = NULL, nscor = 1, alpha = 0.05, power = .80, type = d$type[i])$f*2
    d$p95[i] <- wp.rmanova(d$n[i], d$ng[i], d$nm[i], f = NULL, nscor = 1, alpha = 0.05, power = .95, type = d$type[i])$f*2
    #the *2 is to convert effect sizes of type "f" to effect sizes in Cohen's d
    
  # as for above for independent sample t-tests.  
  # floor() rounds down and ceiling() rounds up for studies with an uneven number of participants
  } else if (d$test[i] == "ttest2") {
    d$small[i] <- wp.t(n1 = floor(d$n[i]/2), n2 = ceiling(d$n[i]/2), d = .2, alpha = 0.05, power = NULL, type = d$type[i], alternative = d$tails[i])$power
    d$medium[i] <- wp.t(n1 = floor(d$n[i]/2), n2 = ceiling(d$n[i]/2), d = .5, alpha = 0.05, power = NULL, type = d$type[i], alternative = d$tails[i])$power
    d$large[i] <- wp.t(n1 = floor(d$n[i]/2), n2 = ceiling(d$n[i]/2), d = .8, alpha = 0.05, power = NULL, type = d$type[i], alternative = d$tails[i])$power
    d$p80[i] <- wp.t(n1 = floor(d$n[i]/2), n2 = ceiling(d$n[i]/2), d = NULL, alpha = 0.05, power = .80, type = d$type[i], alternative = d$tails[i])$d
    d$p95[i] <- wp.t(n1 = floor(d$n[i]/2), n2 = ceiling(d$n[i]/2), d = NULL, alpha = 0.05, power = .95, type = d$type[i], alternative = d$tails[i])$d
    
  # as for above for one sample t-tests and paired sample t-tests
  } else if (d$test[i] == "ttest1"){
    d$small[i] <- wp.t(n1 = d$n[i], n2 = NULL, d = .2, alpha = 0.05, power = NULL, type = d$type[i], alternative = d$tails[i])$power
    d$medium[i] <- wp.t(n1 = d$n[i], n2 = NULL, d = .5, alpha = 0.05, power = NULL, type = d$type[i], alternative = d$tails[i])$power
    d$large[i] <- wp.t(n1 = d$n[i], n2 = NULL, d = .8, alpha = 0.05, power = NULL, type = d$type[i], alternative = d$tails[i])$power
    d$p80[i] <- wp.t(n1 = d$n[i], n2 = NULL, d = NULL, alpha = 0.05, power = .80, type = d$type[i], alternative = d$tails[i])$d
    d$p95[i] <- wp.t(n1 = d$n[i], n2 = NULL, d = NULL, alpha = 0.05, power = .95, type = d$type[i], alternative = d$tails[i])$d
    
  # as for above for correlations, using d = 2*r / sqrt(1 - r^2)
  } else if (d$test[i] == "corr"){
    d$small[i] <- wp.correlation(n = d$n[i], r = 0.1, power = NULL, alpha = 0.05, alternative = d$tails[i])$power
    d$medium[i] <- wp.correlation(n = d$n[i], r = 0.243, power = NULL, alpha = 0.05, alternative = d$tails[i])$power
    d$large[i] <- wp.correlation(n = d$n[i], r = 0.371, power = NULL, alpha = 0.05, alternative = d$tails[i])$power
    r <- wp.correlation(n = d$n[i], r = NULL, power = .80, alpha = 0.05, alternative = d$tails[i])$r
    d$p80[i] <- 2*r / sqrt(1 - r^2)
    r <- wp.correlation(n = d$n[i], r = NULL, power = .95, alpha = 0.05, alternative = d$tails[i])$r
    d$p95[i] <- 2*r / sqrt(1 - r^2)
    
    
  # as for above for linear regression, using f2 small medium and large effect size according to Cohen 1992
  } else if (d$test[i] == "regression"){
    d$small[i] <- wp.regression(n = d$n[i], p1 = 1, f2 = 0.1^2, power = NULL, alpha = 0.05)$power
    d$medium[i] <- wp.regression(n = d$n[i], p1 = 1, f2 = 0.25^2, power = NULL, alpha = 0.05)$power
    d$large[i] <- wp.regression(n = d$n[i], p1 = 1, f2 = 0.4^2, power = NULL, alpha = 0.05)$power
    f2 <- wp.regression(n = d$n[i], p1 = 1, f2 = NULL, power = .80, alpha = 0.05)$f2
    d$p80[i] <- sqrt(f2) * 2 
    f2 <- wp.regression(n = d$n[i], p1 = 1, f2 = NULL, power = .95, alpha = 0.05)$f2
    d$p95[i] <-  sqrt(f2) * 2
    
    
  } else {
    warning(paste("The test for", d$Publication, "cannot be identified"))
  }
}

# return the dataframe with all the powers and sensitivities filled in
  return(d)
}
