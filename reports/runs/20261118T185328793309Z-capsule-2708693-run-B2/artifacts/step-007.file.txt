df$discPro <- ifelse(df$discPro == "Yes, and they present a set of results that includes only studies that were PROSPECTIVELY registered" & df$disc == "Yes", "both", 
                  ifelse(df$discPro == "Yes, and they present a set of results that includes only studies that were PROSPECTIVELY registered" & df$disc == "No", "pro",
                         ifelse(df$discPro == "Yes, but they present results in such a way that we CANNOT SEPARATE out the results for only the studies that were PROSPECTIVELY registered", "unclear", "no")
                  )
)

# recode whether each article reports NON-outcome discrepancies. (This is the same code as above but for non-outcome discrepancies).
df$discProOther <- ifelse(df$discProOther == "Yes, and they present a set of results that includes only studies that were PROSPECTIVELY registered" & df$discOther == "Yes", "both", 
                  ifelse((df$discProOther == "Yes, and they present a set of results that includes only studies that were PROSPECTIVELY registered" & df$discOther == "No"), "pro",
                         ifelse(df$discProOther == "Yes, but they present results in such a way that we CANNOT SEPARATE out the results for only the studies that were PROSPECTIVELY registered", "unclear", "no")
                  )
)

# remove the columns that are made redundant due to the previous two sections of code.
df <- df %>% select(-c(disc, discOther, discProOtherDenom, discOtherDenom))
# rename the "pro" columns. I only assigned the new columns to "pro" so they would keep their position before the related numeric data in the dataframe.
df <- rename(df, discOutcome = discPro)
df <- rename(df, discOther = discProOther)

# simplify the coding for $source
df$source <- gsub("Journal\\(s\\)", "journal", df$source)
df$source <- gsub("Registry\\(ies\\)", "reg", df$source)
df$source <- gsub("Search engine\\(s\\) \\(e.g., PubMed, EMBASE\\)", "searchEngine", df$source)

# divide our potential exploratory analysis column ($explore) into four binary columns.
df$expTotalDisc <- ifelse(grepl("total number", df$explore), 1, 0)
df$expFund <- ifelse(grepl("FUNDING", df$explore), 1, 0)
df$expStatSig <- ifelse(grepl("STATISTICAL SIGNIFICANCE", df$explore), 1, 0)
df$expTime <- ifelse(grepl("TIME OF REGISTRATION", df$explore), 1, 0)
df <- df %>% select(-explore)
