```{r function_metaDisc}
# For running meta-analyses on proportions (most of the meta analyses in this review).

metaDisc <- function(df){
  # initialize a dataframe to hold all 40 meta-analytic estimates, CIs, PIs, and number of articles analyzed.
  estim <- data.frame(matrix(ncol = 7, nrow = length(df)))
  colnames(estim) <- c("b", "lower.random", "upper.random", "lower.predict", "upper.predict", "k", "n")
  
  # assign the colnames from df to the rownames for estim so that the results line up in the next section of code
  rownames(estim) <- colnames(df)
  
  # initialize lists of lists to hold the results of the 40 meta-analyses
  metaList<-list()
  
  # calculate the meta-analytic estimates, CIs, PIs, and number of articles analyzed for outcomes from prospectively registered studies
  for (i in grep("discProAny", colnames(df)):grep("discProTime", colnames(df))){
    if (sum(!is.na(df[,i] > 0 ))){
      # create temporary dataframe with the NAs removed. This is necessary to align the article author and year with the correct article after running each meta-analysis
      dfMeta <- df %>% filter(!is.na(df[,i]))
      meta <- meta::metaprop(dfMeta[,i], dfMeta$discProDenom,
                          # method for conducting the meta-analytic model (a multi-level regression)
                          method="GLMM",
                          # method to calculate tau
                          method.tau="ML",
                          # method to calculate the confidence interval for tau^2 (which we don't report)
                          method.tau.ci="J", 
                          comb.random=TRUE, 
                          prediction=TRUE,
                          level.predict=0.95,
                          # helpful for meta-analyses of few articles (modifies Student's T-distribution)
                          hakn=TRUE
      )
      # create the study labels for the forest plots
      meta$studlab <- paste(dfMeta$author, dfMeta$pubYear)
      
    metaList[[i]] <- meta #store a list of the meta variable for all 40 meta-analyses to make forest plots later
    # All meta-analytic estimates, CIs, and PIs, need to have inv.logit() run on them to reverse the logit() function that is used in the metaprop() function. 
    estim[i,] <- c(inv.logit(meta$TE.random), 
                   inv.logit(meta$lower.random), 
                   inv.logit(meta$upper.random),
                   inv.logit(meta$lower.predict),
                   inv.logit(meta$upper.predict),
                   meta$k,
                   sum(meta$data[,2]))
        }
  }
  
  # same as above for outcomes from studies with unclear timing of registration.
  for (i in grep("discAny", colnames(df)):grep("discTime", colnames(df))){
    if (sum(!is.na(df[,i] > 0 ))){
      
      # create temporary dataframe with the NAs removed
      dfMeta <- df %>% filter(!is.na(df[,i]))
      meta <- meta::metaprop(dfMeta[,i], dfMeta$discDenom,
                             method="GLMM", 
                             method.tau="ML", 
                             method.tau.ci="J", 
                             comb.random=TRUE, 
                             prediction=TRUE,
                             level.predict=0.95,
                             hakn=TRUE
      )
      meta$studlab <- paste(dfMeta$author, dfMeta$pubYear)
      
      metaList[[i]] <- meta # store a list of the meta variable for all 40 meta-analyses to make forest plots later
      estim[i,] <- c(inv.logit(meta$TE.random), 
                     inv.logit(meta$lower.random), 
                     inv.logit(meta$upper.random),
                     inv.logit(meta$lower.predict),
                     inv.logit(meta$upper.predict),
                     meta$k,
                     sum(meta$data[,2]))
    }
  }
      

  # same as above for non-outcome discrepancies for both prospectively and unclear timing of registration articles
  # Increment by 2 because we have a denominator for each non-outcome discrepancy in the column after the numerator. This is also the reason this block of code needs to be separate from the outcome discrepancies, which all share a denominator.
  for (i in seq(from = grep("discProOtherElig", colnames(df))[1], to = grep("discOtherResults", colnames(df))[1], by = 2)) {
    if (sum(!is.na(df[,i] > 0 ))){
      # create temporary dataframe with the NAs removed
      dfMeta <- df %>% filter(!is.na(df[,i]))
      meta <- meta::metaprop(dfMeta[,i], dfMeta[,i+1],
                             method="GLMM", 
                             method.tau="ML", 
                             method.tau.ci="J", 
                             comb.random=TRUE, 
                             prediction=TRUE,
                             level.predict=0.95,
                             hakn=TRUE
      )
      meta$studlab <- paste(dfMeta$author, dfMeta$pubYear)
      
      metaList[[i]] <- meta # store a list of the meta variable for all 40 meta-analyses to make forest plots later
      estim[i,] <- c(inv.logit(meta$TE.random), 
                     inv.logit(meta$lower.random), 
                     inv.logit(meta$upper.random),
                     inv.logit(meta$lower.predict),
                     inv.logit(meta$upper.predict),
                     meta$k,
                     sum(meta$data[,2]))
    }
  }
      

  return(list(estim, metaList))
}
```

```{r function_metaRR}
# For running meta-analyses on Risk Ratios (i.e., likelihood ratios). For the exploratory analyses.

metaRR <- function(Ee, Ne, Ec, Nc, df){
  
  
  meta <- meta::metabin(Ee, Ne, Ec, Nc,
                       # Risk ratio analysis will not take the parameters below that I used in the other analyses. Instead, they default to method = "MH" and method.tau = "DL"
                       # method="GLMM", 
                       # method.tau="ML", 
                       # method.tau.ci="J", 
                        comb.random=TRUE, 
                        prediction=TRUE,
                        level.predict=0.95,
                        hakn=TRUE,
                        sm = "RR"
  )
  
    meta$studlab <- paste(df$author, df$pubYear)
    
    CI <- paste0(round(exp(meta$lower.random),2), "-", round(exp(meta$upper.random),2))  
    
    #sprintf() is necessary to keep trailing zeros on a PI that = 1.00
    PI <- paste0(round(exp(meta$lower.predict),2), "-", sprintf("%.2f", round(exp(meta$upper.predict),3)))
               
                 
  return(list(meta, CI, PI))
}
```

```{r function_makeCI}
# To format the confidence intervals in the tables.
makeCI <- function(data){

  data <- round(data, 2)
  CI <- paste0(data$lower.random*100, "-", data$upper.random*100, "%")
  return(CI)
  
  }
```
