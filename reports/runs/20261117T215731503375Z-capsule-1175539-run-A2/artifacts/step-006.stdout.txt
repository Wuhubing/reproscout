#### anova of cardiac factors over diagnosis ### 
cat("\n\nCalculate Anova of cardiac factors over diagnosis - p values not adjusted for multiple testing\n")
lapply(d[,c("LVEDVi","LVESVi","LVSVi","LVMi","RVEDVi","RVESVi","RVSVi","LVEF","RVEF","concentricity","septalthhickness","PWV")], 
       function(x) summary(aov(x ~ d$dx)))
                                   
### calculate effect sizes: ##
cat("\n\nCalculate effect sizes of cardiac factors over diagnosis\n")
lapply(d[,c("LVEDVi","LVESVi","LVSVi","LVMi","RVEDVi","RVESVi","RVSVi","LVEF","RVEF","concentricity","septalthhickness","PWV")], 
       function(x) cohen.d(x, d$dx))
       

### Compute adjusted models: ##
cat("\n\nCalculate effect of diagnosis over cardiac factors - adjusted for physical activity score - p values not adjusted for multiple testing\n")
res<-lapply(d[,c("LVEDVi","LVESVi","LVSVi","LVMi","RVEDVi","RVESVi","RVSVi","LVEF","RVEF","concentricity","septalthhickness","PWV")],
        function(x) glm(x ~ d$dx  + d$Hvq_Activityscore ))
vars<-names(d[,c("LVEDVi","LVESVi","LVSVi","LVMi","RVEDVi","RVESVi","RVSVi","LVEF","RVEF","concentricity","septalthhickness","PWV")])
for (i in 1:12){print (vars[i])
  print(summary(res[[i]]))
}
